队列：栈是倒叙的回忆，而队列是顺叙的一生

在数据结构中，人们总是把栈和队列放在一起比较

相对于栈给人见面时的陌生感，队列本身的含义则能够让人们更容易理解其含义

如同排队时的进出关系，先进先出，便是队列的含义

同样于栈，他并不是一种底层的数据结构，不同于栈只注重栈顶，队列需要同时考虑前端和后端

队列的实现方式同样也有很多种

第一种是我自己想出来的数组法


#include 
#include 
#define MAX_NUM 100

struct queue{
    int arr[MAX_NUM];
    int num;
};

int InitialQueue(struct queue* pointer){
    pointer->num = -1;
    return pointer->num;
}

_Bool IsFull(struct queue* pointer){
    return pointer->num == MAX_NUM - 1;
}

_Bool IsEmpty(struct queue* pointer){
    return pointer->num == -1;
}

void EnterQueue(struct queue* pointer,int value){
    if(IsFull(pointer)){
        printf("队列已满，无法加入\n");
        return;
    }
    pointer->arr[++pointer->num] = value;
    printf("入列：%d\n",value);
    
}

void ExitQueue(struct queue* pointer){
    if(IsEmpty(pointer)){
        printf("队列已空，无法出列\n");
        return;
    }
    int i = 1;
    printf("出列：%d\n",pointer->arr[0]);
    for(i = 1;i <= pointer->num;i++){
        pointer->arr[i-1] = pointer->arr[i];
    }
    pointer->num--;
    
}

这种队列可以避免假溢出的情况，但是出列时需要数组的每个值向前移动，O(n)的时间复杂度，存在更好的替代方式


含前后索引的数组法：


#include 
#include 

#define MAX 100

struct Queue {
    int items[MAX];
    int front;   /* 队首指针：指向第一个元素 */
    int rear;    /* 队尾指针：指向下一个待插入位置 */
};

void initQueue(struct Queue* q) {
    q->front = 0;
    q->rear = 0;
}

bool isEmpty(struct Queue* q) {
    return q->front == q->rear;
}

bool isFull(struct Queue* q) {
    return q->rear == MAX;
}

/* 入队：在队尾添加元素 */
void enqueue(struct Queue* q, int value) {
    if (isFull(q)) {
        printf("队列已满！\n");
        return;
    }
    q->items[q->rear++] = value;  /* 在 rear 处写入，然后 rear 后移 */
    printf("入队: %d\n", value);
}

/* 出队：从队首移除元素 */
int dequeue(struct Queue* q) {
    if (isEmpty(q)) {
        printf("队列为空！\n");
        return -1;
    }
    return q->items[q->front++];  /* 返回 front 处元素，然后 front 后移 */
}

int main() {
    struct Queue q;
    initQueue(&q);

    enqueue(&q, 10);  /* 入队: 10 */
    enqueue(&q, 20);  /* 入队: 20 */
    enqueue(&q, 30);  /* 入队: 30 */

    printf("出队: %d\n", dequeue(&q));  /* 输出: 出队: 10 (先进先出) */
    printf("出队: %d\n", dequeue(&q));  /* 输出: 出队: 20 */
    printf("出队: %d\n", dequeue(&q));  /* 输出: 出队: 30 */
    return 0;
}

通过前索引的移动来表示出列，有效避免了O(n)的复杂性

然而，这种方式也不是完美的，前索引只会往后移而不会复原，导致前索引前的位置会再也不发使用，也就是所谓的“假溢出”，最终使得队列的有效长度越来越小


将前面两种代码的结合，引出的便是下面的循环队列法

为了使被假溢出的位置得到重新利用，新的方法选择将后索引超过上限取余数使其返回至下标为0，没错，也就是出现了后索引在前索引之前的情况

这种思路一开始会让人感到一种不可言喻反直觉，就像是队列断开了的样子

但仔细想想确实是没有问题的，高手的思路就是不一样，高山仰止啊


#include 
#define MAX_NUM 5
struct queue{
    int arr[MAX_NUM];
    int left;
    int right;
};

void InitialQueue(struct queue* pointer){
    pointer->left = 0;
    pointer->right = 0;
}

_Bool IsFull(struct queue* pointer){
    return (pointer->right + 1) % MAX_NUM == pointer->left;
}

_Bool IsEmpty(struct queue* pointer){
    return pointer->right == pointer->left;
}

void EnterQueue(struct queue* pointer,int value){
    if(IsFull(pointer)){
        printf("队列已满，无法加入！\n");
        return;
    }
    pointer->arr[pointer->right] = value;
    pointer->right = (pointer->right + 1) % MAX_NUM;
    printf("进入队列%d\n",value);
}

void ExitQueue(struct queue* pointer){
    if(IsEmpty(pointer)){
        printf("队列已空，无法退出！\n");
        return;
    }
    printf("退出队列%d\n",pointer->arr[pointer->left]);
    pointer->left = (pointer->left + 1)% MAX_NUM;
}
