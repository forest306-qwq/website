双向链表：创造，插入，遍历，删除

在昨天链表学习的基础上，今天学习了双向链表以及其使用

相对于单链表，双向链表增加了每个节点的prev指针，从而指向上一个节点

struct dllNode{
    int data;
    struct dllNode* prev;
    struct dllNode* next;
};

struct dllNode* dllCreate(int value){
    struct dllNode* NewNode = malloc(sizeof(struct dllNode));
    if(NewNode == NULL){
        printf("内存申请失败\n");
        return NULL;
    }
    NewNode->data = value;
    NewNode->next = NULL;
    NewNode->prev = NULL;
    return NewNode;
}

struct dllNode* InsertHead(struct dllNode* head,struct dllNode* NewNode){
    if(head == NULL){
        printf("原链表为空，无法头部插入\n");
        return NULL;
    }
    if(NewNode == NULL){
        printf("未接收到插入的节点指针\n");
        return NULL;
    }
    NewNode->next = head;
    head->prev = NewNode;
    head = NewNode;
    return head;
}

void InsertAfter(struct dllNode* prevNode,struct dllNode* NewNode){
    if(prevNode == NULL){
        printf("未找到前置节点指针，无法插入\n");
        return;
    }
    if(NewNode == NULL){
        printf("未接收到插入的节点指针\n");
        return;
    }
    if(prevNode -> next != NULL){
        NewNode -> next = prevNode -> next;
        prevNode -> next ->prev = NewNode;
    }
    prevNode -> next = NewNode;
    NewNode -> prev = prevNode;


    return;

}

struct dllNode* InsertTail(struct dllNode* tail,struct dllNode* NewNode){
    if(tail == NULL){
        printf("未找到前置节点指针，无法插入\n");
        return;
    }
    if(NewNode == NULL){
        printf("未接收到插入的节点指针\n");
        return;
    }
    tail -> next = NewNode;
    NewNode -> prev = tail;
    tail = NewNode;
    return tail;
}

struct dllNode* TraveserForward(struct dllNode* head){
    if(head == NULL){
        printf("链表为空\n");
        return NULL;
    }
    int count = 1;
    struct dllNode* traveser = head;
    while(traveser->next != NULL){
        traveser = traveser -> next;
        count++;
    }
    struct dllNode* tail = traveser;
    printf("一共遍历了%d个节点\n",count);
    return tail;
}

struct dllNode* TraveserBackward(struct dllNode* tail){
    if(tail == NULL){
        printf("链表为空\n");
        return NULL;
    }
    int count = 1;
    struct dllNode* traveser = tail;
    while(traveser -> prev != NULL){
        traveser = traveser -> prev;
        count++;
    }
    struct dllNode* head = traveser;
    printf("一共反向遍历了%d个节点\n",count);
    return head;
}

struct dllNode* NodeDelete(struct dllNode* head,struct dllNode* del){
    if(del == head){
        head = del->next;
    }
    if(del ->next != NULL){
        del -> next -> prev = del ->prev;
    }
    if(del ->prev != NULL){
        del ->prev ->next = del -> next;
    }

    free(del);
    return head;
}

各函数功能

    dllCreate:申请内存并创建新节点，赋值，并将prev和next设置为空指针
    InsertHead:通过NewNode->next = head和head->prev = NewNode使得新建的节点作为头部插入原链表头部，并返回新的head
    InsertAfter:需要插入位置的前一个节点的指针（当然非末尾的插入后一个节点的指针也可以）,完成非头部的插入
    InsertTail:需要原尾部指针来完成尾部的插入，尾部指针可以由头部指针正向遍历返回
    TraveserForward:正向遍历，由头部指针得到尾部指针
    TraveserBackward:逆向遍历，由尾部指针得到头部指针
    NodeDelete:删除节点，只需要所删除的节点的指针，以及通过判断前后节点是否存在来决定是连接还是赋值NULL作为头部或尾部

相较于单链表的优点

    单链表只能由前节点的指针指向后节点而不能倒退，意味这前节点的指针的价值一定比后指针更大，而双向链表就解决了这个问题
    双相链表支持逆向遍历，可以逆向打印各节点内容，更具有灵活性
    双向链表节点的删除不需要该节点的前指针，只需要指向自己的指针即可（因为可以推得）
